Without `alt`, the title is used, falling back to the page title. {% endcomment %} {%- assign gallery_folder = include.folder | append: "/" | replace: "//", "/" -%} {%- assign gallery_key = include.folder | split: "/" | last -%} {%- if include.data -%}{%- assign gallery_key = include.data -%}{%- endif -%} {%- assign gallery_data = site.data.photography[gallery_key] -%}
